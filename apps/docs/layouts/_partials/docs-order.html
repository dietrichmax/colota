{{- /* Returns (dict "docs" groups "integrations" groups), each group (dict "label" "id" "pages"), failing the build on a missing or unlisted page. */ -}} {{- $result := dict }} {{- $listed := slice }} {{- range $name := slice "docs" "integrations" }} {{- $groups := slice }} {{- range index hugo.Data.sidebar $name }} {{- $pages := slice }} {{- range .pages }} {{- $p := site.GetPage (printf "/docs/%s" .) }} {{- if not $p }}{{ errorf "data/sidebar.yaml lists %q, but docs/%s.md does not exist" . . }}{{ end }} {{- if in $listed $p.RelPermalink }}{{ errorf "data/sidebar.yaml lists %q twice" . }}{{ end }} {{- if not $p.Description }}{{ errorf "docs/%s.md has no description; the /docs/ hub prints it under the title" . }}{{ end }} {{- $pages = $pages | append $p }} {{- $listed = $listed | append $p.RelPermalink }} {{- end }} {{- $groups = $groups | append (dict "label" .label "id" (anchorize .label) "pages" $pages) }} {{- end }} {{- $result = merge $result (dict $name $groups) }} {{- end }} {{- range (site.GetPage "/docs").RegularPagesRecursive }} {{- if not (in $listed .RelPermalink) }}{{ errorf "%s is not listed in data/sidebar.yaml" .File.Path }}{{ end }} {{- end }} {{- return $result }}
